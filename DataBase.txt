-- =====================================================================
-- Base de datos: auth_db  |  Módulo de autenticación (login y registro)
-- Motor: MySQL 8 (también funciona en MySQL 5.7 y MariaDB)
-- Cubre: HU-01 registro, HU-02 login, HU-03 recuperar contraseña,
--        HU-04 cerrar sesión, HU-05 límite de intentos fallidos
-- =====================================================================
 
DROP DATABASE IF EXISTS auth_db;
 
CREATE DATABASE auth_db
    CHARACTER SET utf8mb4
    COLLATE utf8mb4_unicode_ci;
USE auth_db;
-- ---------------------------------------------------------------------
-- 1) users: las personas registradas (HU-01, HU-02)
-- ---------------------------------------------------------------------
CREATE TABLE users (
    id            BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    name          VARCHAR(100)    NOT NULL,
    email         VARCHAR(255)    NOT NULL,
    -- Hash (bcrypt o argon2), NUNCA la contraseña en texto plano (CA-01.7).
    -- 255 caracteres alcanzan para cualquier algoritmo de hashing común.
    password_hash VARCHAR(255)    NOT NULL,
    created_at    TIMESTAMP       NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at    TIMESTAMP       NOT NULL DEFAULT CURRENT_TIMESTAMP
                                  ON UPDATE CURRENT_TIMESTAMP,
    PRIMARY KEY (id),
    -- La collation utf8mb4_unicode_ci NO distingue mayúsculas de minúsculas,
    -- así que ana@test.local y ANA@test.local cuentan como el mismo correo
    -- (CA-01.3, caso de prueba CP-04).
    UNIQUE KEY uq_users_email (email)
) ENGINE=InnoDB;
 
-- ---------------------------------------------------------------------
-- 2) password_resets: enlaces de recuperación (HU-03)
-- ---------------------------------------------------------------------
CREATE TABLE password_resets (
    id         BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    user_id    BIGINT UNSIGNED NOT NULL,
    -- Se guarda el HASH (SHA-256 en hexadecimal = 64 caracteres) del token,
    -- no el token. El token real solo viaja en el enlace del correo.
    token_hash CHAR(64)        NOT NULL,
    created_at TIMESTAMP       NOT NULL DEFAULT CURRENT_TIMESTAMP,
    -- La aplicación guarda aquí created_at + 30 minutos (CA-03.4).
    expires_at DATETIME        NOT NULL,
    -- Se llena al usar el enlace: así es de un solo uso (CA-03.3, CA-03.6).
    used_at    DATETIME        NULL DEFAULT NULL,
    PRIMARY KEY (id),
    UNIQUE KEY uq_password_resets_token (token_hash),
    KEY idx_password_resets_user (user_id),
    CONSTRAINT fk_password_resets_user
        FOREIGN KEY (user_id) REFERENCES users (id)
        ON DELETE CASCADE
) ENGINE=InnoDB;
 
-- ---------------------------------------------------------------------
-- 3) login_attempts: registro de intentos de inicio de sesión (HU-05)
-- ---------------------------------------------------------------------
CREATE TABLE login_attempts (
    id           BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    -- Se guarda el correo escrito (no user_id) para poder limitar también
    -- los intentos contra cuentas que no existen sin revelar si existen.
    email        VARCHAR(255)    NOT NULL,
    ip_address   VARCHAR(45)     NOT NULL,   -- 45 caracteres alcanzan para IPv6
    was_success  TINYINT(1)      NOT NULL DEFAULT 0,
    attempted_at DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (id),
    -- Acelera la consulta "intentos de esta cuenta + IP en los últimos 15 min".
    KEY idx_login_attempts_lookup (email, ip_address, attempted_at)
) ENGINE=InnoDB;
 
-- ---------------------------------------------------------------------
-- 4) sessions: sesiones activas en el servidor (HU-04)
--    Útil si tu aplicación guarda las sesiones en base de datos. Si tu
--    framework ya trae su propia tabla de sesiones, puedes omitir esta.
-- ---------------------------------------------------------------------
CREATE TABLE sessions (
    id            VARCHAR(128)    NOT NULL,
    user_id       BIGINT UNSIGNED NULL,
    ip_address    VARCHAR(45)     NULL,
    user_agent    VARCHAR(255)    NULL,
    created_at    TIMESTAMP       NOT NULL DEFAULT CURRENT_TIMESTAMP,
    last_activity DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (id),
    KEY idx_sessions_user (user_id),
    -- Cerrar sesión = borrar la fila: la sesión queda invalidada (CA-04.2).
    CONSTRAINT fk_sessions_user
        FOREIGN KEY (user_id) REFERENCES users (id)
        ON DELETE CASCADE
) ENGINE=InnoDB;
